class Solution {
    public String dayOfTheWeek(int day, int month, int year) {
        String[] days = {"Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"};
        int[] daysInMonth = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};

        // January 1, 1971 was a Friday (index 5 in our days array)
        int totalDays = 0;

        // Count days from full years from 1971 up to year - 1
        for (int y = 1971; y < year; y++) {
            totalDays += isLeapYear(y) ? 366 : 365;
        }

        // Count days for full months in the current year
        for (int m = 0; m < month - 1; m++) {
            totalDays += daysInMonth[m];
            if (m == 1 && isLeapYear(year)) { // February in a leap year
                totalDays += 1;
            }
        }

        // Add remaining days in current month
        totalDays += day;

        // January 1, 1971 offset: (5 + totalDays - 1) % 7
        return days[(5 + totalDays - 1) % 7];
    }

    private boolean isLeapYear(int year) {
        return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
    }
}
